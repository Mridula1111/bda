wget https://archive.apache.org/dist/sqoop/1.4.7/sqoop-1.4.7.bin__hadoop-2.6.0.tar.gz

sudo cp mysql-connector-j-8.0.33/mysql-connector-j-8.0.33.jar .

sqoop import \
--connect jdbc:mysql://localhost/company \
--username root \
--table employees \
--target-dir /user/ubuntu/employees_data \
--m 1

check: hdfs dfs -cat /user/ubuntu/employees_data/part-m-00000

Create a new table in MySQL for export:
mysql> CREATE TABLE employees_export (id INT, name VARCHAR(50), department VARCHAR(50), salary INT);

in /usr/local/sqoop/lib:
sqoop export \
--connect jdbc:mysql://localhost/company \
--username root \
--table employees_export \
--export-dir /user/ubuntu/employees_filtered \
--input-fields-terminated-by ',' \
--m 1

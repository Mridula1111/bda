nano thrift_test.py

import happybase

# Connect to the HBase Thrift server
conn = happybase.Connection('localhost', port=9090)
conn.open()

# Display existing tables
print(conn.tables())

# Create a table with one column family
conn.create_table('thrift_demo', {'cf1': dict()})

# Access the table
table = conn.table('thrift_demo')

# Insert a row
table.put(b'row1', {b'cf1:name': b'Archana'})

# Retrieve and display the row
print(table.row(b'row1'))

# Scan all rows
for key, data in table.scan():
    print(key, data)

# Disable and delete the table
conn.disable_table('thrift_demo')
conn.delete_table('thrift_demo')

# Close the connection
conn.close()
